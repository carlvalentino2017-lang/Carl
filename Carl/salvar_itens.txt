pi = 3.141592653589793
gravidade = 0
jump = 0.01
jump3 = 1
height = 0

if jump2 == "s":
    jump1 = True
    print(height)
elif jump2 != "s":
    jump1 = False

    height = jump - gravidade

    print(height)
